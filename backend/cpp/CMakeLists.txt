cmake_minimum_required(VERSION 3.15)
project(cpp_recommender LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(ranking recommender.cpp utils.cpp)
target_include_directories(ranking PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
option(BUILD_SERVER "Build the native HTTP server" ON)
if(BUILD_SERVER)
    find_package(Threads REQUIRED)
    find_package(SQLite3 REQUIRED)
    add_library(user_store user_store.cpp)
    if(TARGET SQLite3::SQLite3)
        target_link_libraries(user_store PUBLIC ranking SQLite3::SQLite3)
    else()
        target_link_libraries(user_store PUBLIC ranking SQLite::SQLite3)
    endif()
    add_executable(recommender main.cpp)
    target_link_libraries(recommender PRIVATE user_store Threads::Threads)
    if(WIN32)
        target_link_libraries(recommender PRIVATE ws2_32)
    endif()
endif()
if(BUILD_TESTING)
    if(BUILD_SERVER)
        add_executable(session_revocation_test ../../tests/session_revocation_test.cpp)
        target_link_libraries(session_revocation_test PRIVATE user_store Threads::Threads)
        add_test(NAME session_revocation COMMAND session_revocation_test ${CMAKE_CURRENT_SOURCE_DIR}/../data/catalog.json)
    endif()
    find_package(Threads REQUIRED)
    add_executable(session_admission_test ../../tests/session_admission_test.cpp)
    target_include_directories(session_admission_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(session_admission_test PRIVATE Threads::Threads)
    add_test(NAME session_admission COMMAND session_admission_test)
    add_executable(recommender_test ../../tests/recommender_test.cpp)
    target_link_libraries(recommender_test PRIVATE ranking)
    add_test(NAME ranking_validation COMMAND recommender_test)
endif()

option(BUILD_BENCHMARKS "Build synthetic exact-retrieval benchmark" OFF)
if(BUILD_BENCHMARKS)
    add_executable(retrieval_benchmark ../../benchmarks/retrieval.cpp)
    target_link_libraries(retrieval_benchmark PRIVATE ranking)
endif()
